import java.util.*;

class Solution { 
    // Static inner class so it can be instantiated without an instance of Solution
    static class Node {
        int val; 
        int row; 
        int col; 
        
        public Node(int val, int row, int col) { 
            this.val = val; 
            this.row = row; 
            this.col = col; 
        } 
    } 

   
    static class Compare implements Comparator<Node> { 
        @Override 
        public int compare(Node a, Node b) { 
            return Integer.compare(a.val, b.val); // for the min heap
        } 
    } 

    public ArrayList<Integer> mergeArrays(int[][] mat) { 
        int n = mat.length; 
        int m = mat[0].length; 
        ArrayList<Integer> res = new ArrayList<>(); 
        
       
        PriorityQueue<Node> pq = new PriorityQueue<>(new Compare()); 
        
        // Loop through rows and add the first element of each row
        for(int i = 0; i < n; i++) { 
            // Fixed variable 'mat' and array syntax mat[i][0]
            pq.add(new Node(mat[i][0], i, 0)); 
        } 
        
        while(!pq.isEmpty()) { 
            Node current = pq.poll();  
            res.add(current.val); 
            
            int column = current.col; 
            int row = current.row;
            
            // If there are more elements in the same row, add the next one
            if(column + 1 < m) { 
                pq.add(new Node(mat[row][column + 1], row, column + 1)); 
            } 
        } 
        return res; 
    } 
}
